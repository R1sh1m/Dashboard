cmake_minimum_required(VERSION 3.16)

# Locate valid IDF_PATH (fallback to C:/esp/v6.1/esp-idf if env is unset or pointing to a stale path)
if(DEFINED ENV{IDF_PATH} AND EXISTS "$ENV{IDF_PATH}/tools/cmake/project.cmake")
    file(TO_CMAKE_PATH "$ENV{IDF_PATH}" IDF_PATH)
elseif(EXISTS "C:/esp/v6.1/esp-idf/tools/cmake/project.cmake")
    set(IDF_PATH "C:/esp/v6.1/esp-idf")
elseif(EXISTS "C:/esp/v6.0.3/esp-idf/tools/cmake/project.cmake")
    set(IDF_PATH "C:/esp/v6.0.3/esp-idf")
else()
    message(FATAL_ERROR "IDF_PATH is not set or not found. Set it to the ESP-IDF installation directory.")
endif()

# Ensure ESP_ROM_ELF_DIR is available for gdbinit generation
if(NOT DEFINED ENV{ESP_ROM_ELF_DIR} OR "$ENV{ESP_ROM_ELF_DIR}" STREQUAL "")
    set(_user_home "$ENV{USERPROFILE}")
    if(NOT _user_home)
        set(_user_home "$ENV{HOME}")
    endif()
    if(_user_home)
        file(TO_CMAKE_PATH "${_user_home}/.espressif/tools/esp-rom-elfs/20241011/" _rom_dir)
        if(EXISTS "${_rom_dir}")
            set(ENV{ESP_ROM_ELF_DIR} "${_rom_dir}")
        endif()
    endif()
endif()

# Suppress false-positive component validation warnings (e.g. mutual esp_wifi / wpa_supplicant headers)
set(IDF_COMPONENT_VALIDATION OFF CACHE BOOL "Disable IDF component cross-include validation warnings" FORCE)

include("${IDF_PATH}/tools/cmake/project.cmake")
project(esp_receiver)
